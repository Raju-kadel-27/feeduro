The API Explorer gives you a live, interactive view of every endpoint. Enter your API key, fill in parameters, and send real requests without leaving the docs.

<Aside type="tip">
  Need an API key? Go to **Settings → API Keys** in your Feediyo workspace, or [create one now](https://app.feediyo.com/settings/api-keys).
</Aside>

## Interactive explorer

<div id="scalar-container">
  <p id="scalar-status" style="color: var(--sl-color-gray-3); font-size: 0.875rem; margin: 0;">
    Loading API explorer…
  </p>
</div>

<script type="module">{`
  const container = document.getElementById('scalar-container');
  const status = document.getElementById('scalar-status');

  // Fallback shown if Scalar doesn't load within 6 seconds
  const fallbackTimer = setTimeout(() => {
    if (status) {
      status.innerHTML =
        'API explorer failed to load. ' +
        '<a href="/openapi.yaml" download>Download the OpenAPI spec</a> ' +
        'to import into Postman or Insomnia.';
      status.style.color = 'var(--sl-color-text)';
    }
  }, 6000);

  try {
    const { createScalarApiReference } = await import(
      'https://cdn.jsdelivr.net/npm/@scalar/api-reference@latest/dist/browser/index.js'
    );
    clearTimeout(fallbackTimer);
    status?.remove();
    createScalarApiReference(container, {
      url: '/openapi.yaml',
      configuration: {
        theme: 'default',
        hideModels: false,
        hideDownloadButton: true,
        authentication: { preferredSecurityScheme: 'bearerAuth' },
      },
    });
  } catch {
    clearTimeout(fallbackTimer);
    if (status) {
      status.innerHTML =
        'API explorer failed to load. ' +
        '<a href="/openapi.yaml" download>Download the OpenAPI spec</a> ' +
        'to import into Postman or Insomnia.';
      status.style.color = 'var(--sl-color-text)';
    }
  }
`}</script>
